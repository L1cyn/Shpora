7.3 Шифр Цезаря
сложно
Сдвинь каждую латинскую букву на N позиций. «abc» → 1 → «bcd».
💡 Решение 
def caesar(s, shift):
    result = ""
    for ch in s:
        if ch.isalpha():
            base = ord('A') if ch.isupper() else ord('a')
            result += chr((ord(ch) - base + shift) % 26 + base)
        else:
            result += ch
    return result

print(caesar("Hello, World!", 3))  # Khoor, Zruog!

средне

💡 Решение 
class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.__balance = balance
    def deposit(self, amount):
        self.__balance += amount
    def withdraw(self, amount):
        if amount > self.__balance:
            raise ValueError("Недостаточно средств")
        self.__balance -= amount
    def get_balance(self):
        return self.__balance

acc = BankAccount("Аня", 1000)
acc.deposit(500)
acc.withdraw(300)
print(acc.get_balance())  # 1200
10.4 Магические __str__, __eq__, __len__
средне
Создай класс Book с полями title, pages. Реализуй __str__ (красивый вывод), __eq__ (книги равны при равных названиях), __len__ (возвращает pages).
💡 Решение 
class Book:
    def __init__(self, title, pages):
        self.title = title
        self.pages = pages
    def __str__(self):
        return f'«{self.title}», {self.pages} стр.'
    def __eq__(self, other):
        return self.title == other.title
    def __len__(self):
        return self.pages

b = Book("1984", 328)
print(b)        # «1984», 328 стр.
print(len(b))   # 328
10.5 @property
сложно
Класс Temperature с температурой в Цельсиях. Свойство fahrenheit преобразует к Фаренгейтам.
💡 Решение 
class Temperature:
    def __init__(self, c):
        self.celsius = c
    @property
    def fahrenheit(self):
        return self.celsius * 9/5 + 32
    @fahrenheit.setter
    def fahrenheit(self, f):
        self.celsius = (f - 32) * 5/9

t = Temperature(100)
print(t.fahrenheit)   # 212.0
t.fahrenheit = 32
print(t.celsius)      # 0.0

12.4 Декоратор @timer
сложно
Напиши декоратор, измеряющий время выполнения функции.
💡 Решение 
import time
from functools import wraps

def timer(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        t = time.perf_counter()
        result = func(*args, **kwargs)
        print(f"{func.__name__}: {time.perf_counter()-t:.4f}s")
        return result
    return wrapper

@timer
def slow():
    time.sleep(1)
slow()

import json
from dataclasses import dataclass, asdict, field
from datetime import date
from collections import Counter
from pathlib import Path

FILE = Path("budget.json")

@dataclass
class Transaction:
    type: str           # "income" | "expense"
    category: str
    amount: float
    comment: str = ""
    date: str = field(default_factory=lambda: str(date.today()))

class Budget:
    def __init__(self):
        self.items: list[Transaction] = []
        self.load()

    def add(self, t): self.items.append(t); self.save()

    def balance(self):
        s = sum(t.amount if t.type == "income" else -t.amount
                for t in self.items)
        return s

    def top_categories(self, n=3):
        c = Counter()
        for t in self.items:
            if t.type == "expense":
                c[t.category] += t.amount
        return c.most_common(n)

    def save(self):
        FILE.write_text(json.dumps(
            [asdict(t) for t in self.items],
            ensure_ascii=False, indent=2), encoding="utf-8")

    def load(self):
        if FILE.exists():
            self.items = [Transaction(**d) for d in json.loads(FILE.read_text("utf-8"))]

def menu():
    b = Budget()
    while (cmd := input("\n[1] добавить [2] баланс [3] топ-категорий [0] выход: ")) != "0":
        match cmd:
            case "1":
                t = Transaction(
                    type=input("income/expense: "),
                    category=input("категория: "),
                    amount=float(input("сумма: ")),
                    comment=input("коммент: "),
                )
                b.add(t)
            case "2": print(f"Баланс: {b.balance()}")
            case "3": print(b.top_categories())

if __name__ == "__main__": menu()

import random
from dataclasses import dataclass

@dataclass
class Song:
    title: str
    artist: str
    duration: int   # секунды

class Playlist:
    def __init__(self, name):
        self.name = name
        self.songs: list[Song] = []

    if m := re.match(r"меня зовут (\w+)", msg):
        user_name = m.group(1).capitalize()
        return f"Приятно познакомиться, {user_name}!"

    match msg:
        case "привет" | "здравствуй":
            return f"Привет{', ' + user_name if user_name else ''}!"
        case "как тебя зовут":
            return "Я — Робо-Чат 🤖"
        case "который час" | "время":
            return datetime.now().strftime("%H:%M")
        case "монетка":
            return random.choice(["Орёл", "Решка"])
        case "кубик":
            return str(random.randint(1, 6))
        case "помощь" | "help":
            return "Команды: привет, как тебя зовут, время, монетка, кубик, выход"
        case _:
            return "Не понял. Напиши 'помощь'."

while (msg := input("> ")) != "выход":
    print("🤖", reply(msg))
